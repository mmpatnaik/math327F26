\questionheading{Short questions}{8}
\textbf{A. Short answer (2 points each)}
\begin{enumerate}[label=\textbf{A\arabic*.}]
\item In the additive group $\Z/18\Z$, find the order of $\overline{6}$ and list the elements of $\langle\overline{6}\rangle$.
\par\vspace*{1in}
\item Let $\varphi:\Z\to\Z/8\Z$ be given by $\varphi(n)=\overline{n}$. Describe $\ker\varphi$ and $\im\varphi$.
\par\vspace*{1in}
\end{enumerate}
\textbf{B. Multiple choice (1 point each)}
\begin{enumerate}[label=\textbf{B\arabic*.}]
\item Let $H\leq G$ and $a,b\in G$. Which condition is equivalent to $aH=bH$?

\begin{tabular}{@{}p{0.47\linewidth}p{0.47\linewidth}@{}}
(a) $ab^{-1}\in H$ & (b) $b^{-1}a\in H$ \\[5pt]
(c) $a^{-1}b\notin H$ & (d) $a=b$
\end{tabular}


\item Let $a,b,c\in\Z$ be positive integers. Which of the following statements is \textbf{NOT always true}?

(a) $\gcd(a,b)=ma+nb$ for some integers $m,n\in\Z$.

(b) $a\Z+b\Z+c\Z$ is a cyclic subgroup of $\Z$.

(c) If $\gcd(a,b)=1$, then $a\Z+b\Z=\Z$.

(d) If $a<b$, then $a\Z\supseteq b\Z$.



\end{enumerate}

\questionheading{Question 1}{8}
\begin{enumerate}[label=(\alph*)]
\item \textbf{[2 points]} Let $p=(1\ 4\ 2)(3\ 5)$ and $q=(1\ 2\ 3)$ in $S_5$. Compute $pq$ and express it as a product of \textbf{disjoint cycles}.
\par\vspace*{1.7in}
\item \textbf{[3 points]} Let $\sigma\in S_n$ be a cycle $\sigma=(i_1\ i_2\ \cdots\ i_k)$ with $k$ odd. What is the \textbf{order} of $\sigma$, and is $\sigma$ an \textbf{even or an odd} permutation? Justify briefly.
\par\vspace*{2in}
\item \textbf{[3 points]} Let $a$ and $b$ be elements of a group $G$. Suppose $a$ has order $7$ and $a^3b=ba^3$. Show that \[ ab=ba. \]


 
\end{enumerate}

\questionheading{Question 2}{8}
Let $H$ and $K$ be subgroups of a group $G$.
\begin{enumerate}[label=(\alph*)]
\item \textbf{[3 points]} Prove that $H\cap K$ is a subgroup of $G$.
\par\vspace*{2.2in}

\item \textbf{[2 points]} Give an example of a group $G$ and subgroups $H,K\leq G$ for which $H\cup K$ is not a subgroup of $G$. Specify $G$, $H$, and $K$ explicitly.
\par\vspace*{1.3in}

\item \textbf{[3 points]} Let $\varphi: G \rightarrow G'$ be a homomorphism of groups. Show that $\ker(\varphi)$ is a normal subgroup of $G$.

\end{enumerate}

\questionheading{Question 3}{8}
Let $p$ and $q$ be prime numbers with $p < q$, and let $G$ be a group of order $pq$. Show that $G$ has an element of order $p$.

\textit{Do not use Cauchy's theorem or another theorem that directly implies the result.}
\begin{enumerate}[label=(\alph*)]
\item \textbf{[2 points]} Let $x\in G$ with $x\neq 1$. Show that $\operatorname{ord}(x)\in\{p,q,pq\}$.
\par\vspace*{1.5in}
\item \textbf{[2 points]} If some element has order $pq$, show that $G$ has an element of order $p$.
\par\vspace*{1.3in}
\item \textbf{[4 points]} Suppose every non-identity element of $G$ has order $q$. Show that two distinct subgroups of order $q$ intersect only in $\{1\}$, and count elements to reach a contradiction. Conclude that $G$ has an element of order $p$.
\end{enumerate}
