\section{Review of ideals and principal ideal domains}

Recall from Lecture 27 that all rings here are commutative with identity.

\begin{define}[Ideal]
A subset $I\subseteq R$ is an \emph{ideal} if
\begin{enumerate}[label=(\arabic*)]
  \item $0\in I$;
  \item if $a,b\in I$, then $a+b\in I$;
  \item if $a\in I$ and $b\in R$, then $ab\in I$.
\end{enumerate}
\end{define}

Conditions (1) and (2), together with (3) applied to $b=-1$, give
that every ideal is an additive subgroup of $R$.

\begin{define}[Principal ideal]
An ideal $I$ is \emph{principal} if it is generated by a single
element $f\in R$, so $I=(f)=\{bf:b\in R\}$.
\end{define}

\begin{define}[Principal ideal domain]
A \emph{principal ideal domain} (PID) is an $R$ so that there are no elements $a, b \neq 0$ such that $ab=0$ (i.e. $R$ is a domain) and in which
every ideal is principal.
\end{define}

\begin{theorem}
If $F$ is a field, then $F[x]$ is a principal ideal domain.
\end{theorem}

\begin{proof}
Since $F$ is a field, leading coefficients of nonzero polynomials
have nonzero product, so $F[x]$ is an integral domain.

Let $I\subseteq F[x]$ be any ideal. If $I=(0)$, it is principal.
If $I\neq(0)$, then $I$ contains a nonzero polynomial of smallest
degree; call it $f\in I$, say
\[
  f(x)=a_nx^n+a_{n-1}x^{n-1}+\cdots+a_0,\qquad a_n\neq0.
\]
Since $F$ is a field and $f\in I$, so is $\frac1{a_n}f\in I$.
That is, there exists a monic polynomial
\[
  x^n+b_{n-1}x^{n-1}+\cdots+b_0\in I
\]
of smallest degree; replace $f$ by it.

Now let $g(x)\in I$. By the division algorithm for $F[x]$, since $f$
is monic, we may write
\[
  g(x)=h(x)f(x)+r(x)
\]
for some $r(x)$ which is either $0$ or of degree less than
$\deg(f)$. Since $g\in I$ and $hf\in I$,
\[
  g-hf=r\in I.
\]
If $r\neq0$, then $\deg(r)<\deg(f)$, contradicting the choice of $f$
as an element of smallest degree in $I$. Therefore $r=0$ and
$g=h\cdot f$, i.e.\ $g\in(f)$. Hence $I\subseteq(f)$. Since $f\in I$,
also $(f)\subseteq I$, so $I=(f)$.
\end{proof}

\begin{example}[A non-principal ideal of {$\Z[x]$}]
Consider the ideal
\[
  I=(2,x)=\{2a(x)+xb(x):a(x),b(x)\in\Z[x]\}.
\]
Suppose $I=(g)$ for some $g\in\Z[x]$. Since $g$ divides $2$,
it is a nonzero constant, so $g=\pm1$ or $g=\pm2$. Since $g$
also divides $x$ and the coefficient of $x$ is $1$, we must have
$g=\pm1$. This would give $I=\Z[x]$.

But the constant term of every element of $I$ is even, so $1\notin I$.
Thus $(2,x)$ is not principal.
\end{example}

\section{A kernel in \texorpdfstring{$\Q[x]$}{Q[x]}}

Since $\Q$ is a field, every ideal of $\Q[x]$ is principal.

\begin{example}[The kernel of evaluation at {$\sqrt[3]{2}$}]
Let $\gamma=\sqrt[3]{2}$ be the real cube root of $2$. Recall the
evaluation homomorphism from Lecture 27.
\[
  \Phi:\Q[x]\to\C,\qquad x\mapsto\gamma.
\]
What is $\ker(\Phi)$? Is it $0$? We know $\ker(\Phi)=(k)$ for
some $k\in\Q[x]$. First, $x^3-2\in\ker(\Phi)$, since
\[
  \Phi(x^3-2)=\gamma^3-2=0.
\]
If $x^3-2\in\ker(\Phi)$, then $g\cdot(x^3-2)\in\ker(\Phi)$
for every $g\in\Q[x]$, so $(x^3-2)\subseteq\ker(\Phi)$. But is
there any element in $\ker(\Phi)$ of lower positive degree?

Just as in the correspondence $F[x]\longleftrightarrow\Z$, where
\[
  n\mid m\iff(m)\subseteq(n),
\]
we have for polynomials
\[
  f\mid g\iff(g)\subseteq(f).
\]
So
\[
  (x^3-2)\subseteq\ker(\Phi)=(k)\iff k\mid x^3-2,\quad k\in\Q[x].
\]
The polynomial $k$ is not a nonzero constant, since then
$\ker(\Phi)=\Q[x]$, whereas $\Phi(1)=1\neq0$.

If $x^3-2$ were a product of two positive-degree polynomials in
$\Q[x]$, their degrees would add to $3$, so one factor would be
linear. Its root would be a rational root of $x^3-2$. Suppose
$p/q$ is such a root, with $p,q\in\Z$, $q>0$, and $\gcd(p,q)=1$.
Then $p^3=2q^3$, so $p$ is even. Writing $p=2m$ gives
$q^3=4m^3$, so $q$ is even as well. This contradicts $\gcd(p,q)=1$.

Thus $x^3-2$ is irreducible over $\Q$. Since $k$ is a nonconstant
divisor of $x^3-2$, it is a
nonzero scalar multiple of $x^3-2$ and
\[
  \ker(\Phi)=(x^3-2).
\]
\end{example}

\section{Quotient rings}

Let $R$ be a ring and $I\subseteq R$ an ideal. Then we can form a
new ring $R/I$, which consists of the cosets
\[
  a+I,\qquad a\in R.
\]
As an additive group, this is the quotient group of $R$ by the
subgroup $I$. What is the multiplication? We define
\[
  (a+I)\cdot(b+I)\coloneqq ab+I.
\]

\begin{proposition}\label{prop:well-defined}
This gives a well-defined multiplication on $R/I$.
\end{proposition}
\begin{proof}
Suppose $a+I=\tilde a+I$ and $b+I=\tilde b+I$, i.e.\
$a-\tilde a\in I$ and $b-\tilde b\in I$. Then
\[
  (a+I)(b+I)=ab+I,\qquad(\tilde a+I)(\tilde b+I)=\tilde a\tilde b+I,
\]
and we must show $ab+I=\tilde a\tilde b+I$, i.e.\
$ab-\tilde a\tilde b\in I$. Why does this hold? We have
\[
  \tilde b(a-\tilde a)+a(b-\tilde b)=ab-\tilde a\tilde b,
\]
and both terms on the left lie in $I$ because an ideal is closed under
multiplication by elements of $R$. Since $I$ is closed under addition,
$ab-\tilde a\tilde b\in I$.
\end{proof}

\begin{exercise}
Check that $R/I$ is a ring with
\begin{align*}
  \text{addition: }&(a+I)+(b+I)=(a+b)+I,\\
  \text{multiplication: }&(a+I)(b+I)=ab+I,
\end{align*}
zero element $0+I$, and identity element $1+I$.
\end{exercise}

\begin{define}[Quotient ring]
The ring $R/I$ with the operations above is the \emph{quotient ring}
of $R$ by the ideal $I$.
\end{define}

The map
\[
  \pi:R\to R/I,\qquad a\mapsto a+I,
\]
is a ring homomorphism, often called the \emph{canonical quotient
map}. Its kernel is exactly $I$.

\begin{example}[{$\Z/n\Z$}]
Take $R=\Z$ and $I=(n)$. Then $R/I=\Z/n\Z$, with multiplication of
residue classes $\bar a\cdot\bar b=\overline{ab}$. For instance, when
$n=12$,
\[
  \bar5\cdot\bar6=\overline{30}=\bar6.
\]
\end{example}
