\section{Polynomial rings}

Throughout, rings are commutative with identity, as in Lecture 25.

\begin{define}[Polynomial ring]
Let $R$ be a ring. The \emph{ring of polynomials over $R$} is
\[
  R[x]=\{f=a_0+a_1x+\cdots+a_nx^n:a_i\in R,\ n\geq0\}.
\]
\end{define}

\begin{define}[Degree, leading coefficient, monic]
Let $f=\sum_i a_ix^i\in R[x]$ be nonzero.
\begin{itemize}
  \item The \emph{degree} $\deg(f)$ is the largest $n$ with
    $a_n\neq0$.
  \item The \emph{leading} (or \emph{highest}) \emph{coefficient}
    of $f$ is $a_n$ for this largest $n$.
  \item $f$ is \emph{monic} if its leading coefficient is $1$.
\end{itemize}
\end{define}

\begin{proposition}
$R[x]$ is a ring under the following operations. For
$f=\sum_ia_ix^i$ and $g=\sum_jb_jx^j$,
\[
  f+g=\sum_i(a_i+b_i)x^i,\qquad
  f\cdot g=\sum_k\Bigl(\sum_{i+j=k}a_ib_j\Bigr)x^k,
\]
where missing coefficients are taken to be $0$. The constant
polynomials $0$ and $1$ are the additive and multiplicative identities.
\end{proposition}

In $\Z$, given $m$ and $n>0$, we can write $m=an+r$ with $0\leq r<n$.
In $\Z[x]$, dividing $7x+1$ by $2x$ with constant remainder would
require the quotient $7/2$. A suitable leading coefficient lets us
divide with remainder in $R[x]$.

\begin{theorem}[Division with remainder]\label{thm:division}
Let $f(x)\in R[x]$ be monic. Then for any $g(x)\in R[x]$ there exist
unique $q(x),r(x)\in R[x]$ such that
\[
  g=q\cdot f+r,
\]
where $r=0$ or $\deg(r)<\deg(f)$.
\end{theorem}

\begin{remark}
Theorem~\ref{thm:division} also holds when the leading coefficient of
$f$ is a unit. Over a field, we can therefore divide with remainder
by any nonzero polynomial, as in $\R[x]$ in Math 30-1.
\end{remark}

\section{Finite fields}

The rings $\R$ and $\C$ are familiar fields. A field can also have
finitely many elements, as the following example shows.

\begin{example}[The finite field $\F_p$]
For a prime $p$, the ring $\F_p=\Z/p\Z$ is a field with $p$ elements.
Its classes of $0$ and $1$ are distinct. If $a\not\equiv0\pmod p$,
then $\gcd(a,p)=1$, so there are integers $m,n$ with
\[
  ma+np=1.
\]
Reducing modulo $p$ gives $ma\equiv1\pmod p$, so the class of $m$ is
the multiplicative inverse of the class of $a$.
\end{example}

\begin{example}[A nonunit in $\Z/6\Z$]
In $\Z/6\Z$ the class of $2$ has no inverse. Indeed, if
$a\cdot2\equiv1\pmod6$, then multiplying by $3$ gives
\[
  a\cdot(2\cdot3)\equiv3\pmod 6.
\]
Since $2\cdot3=6\equiv0$, this says $3\equiv0\pmod6$, a contradiction.
\end{example}

\section{Ring homomorphisms}

\begin{define}[Ring homomorphism]
Let $R$ and $S$ be rings. A map $\varphi:R\to S$ is a
\emph{homomorphism} if, for all $a,b\in R$,
\[
  \varphi(a+b)=\varphi(a)+\varphi(b),\qquad
  \varphi(ab)=\varphi(a)\varphi(b),
\]
and $\varphi(1_R)=1_S$.
\end{define}

As a homomorphism of additive groups, $\varphi$ also satisfies
$\varphi(0_R)=0_S$ and $\varphi(-a)=-\varphi(a)$.

\begin{example}[Homomorphisms out of $\Z$]
For any ring $R$ there is a unique homomorphism $\Z\to R$. It must send
$1\mapsto1$, and therefore for $n\geq0$
\[
  n\longmapsto\underbrace{1+1+\cdots+1}_{n},
\]
while $-n\mapsto-(1+\cdots+1)$ because a homomorphism preserves
additive inverses.
\end{example}

\section{The substitution principle}

A homomorphism $\Phi:R[x]\to S$ \emph{extends} a homomorphism
$\varphi:R\to S$ if $\Phi(r)=\varphi(r)$ for every $r\in R$,
viewed as a constant polynomial.

\begin{proposition}[Substitution principle]\label{prop:substitution}
Let $\varphi:R\to S$ be a ring homomorphism. For each $\alpha\in S$
there is a unique ring homomorphism $\Phi:R[x]\to S$ with
$\Phi(r)=\varphi(r)$ for all $r\in R$ and $\Phi(x)=\alpha$. It is given by
\[
  \Phi\Bigl(\sum_i a_ix^i\Bigr)=\sum_i\varphi(a_i)\alpha^i.
\]
\end{proposition}

We will prove the substitution principle in Lecture 27.
