\section{Review of homomorphisms and substitution}

Throughout, rings are commutative with identity.

\begin{define}[Ring homomorphism]
Let $R$ and $S$ be rings. A map $\varphi:R\to S$ is a
\emph{ring homomorphism} if, for every $a,b\in R$,
\[
  \varphi(a+b)=\varphi(a)+\varphi(b),\qquad
  \varphi(ab)=\varphi(a)\varphi(b),\qquad
  \varphi(1_R)=1_S.
\]
\end{define}

The addition condition also gives $\varphi(0_R)=0_S$ and
$\varphi(-a)=-\varphi(a)$.

In Lecture 26 we stated the substitution principle. We now prove it.

\begin{proposition}[Substitution principle]
Let $\varphi:R\to S$ be a ring homomorphism. For each $\alpha\in S$
there is a unique ring homomorphism $\Phi:R[x]\to S$ with
$\Phi(r)=\varphi(r)$ for all $r\in R$ and $\Phi(x)=\alpha$. It is given by
\[
  \Phi\Bigl(\sum_i a_ix^i\Bigr)=\sum_i\varphi(a_i)\alpha^i.
\]
\end{proposition}
\begin{proof}
Fix $\alpha\in S$. If $\Phi$ extends $\varphi$ and sends $x$ to
$\alpha$, then additivity and multiplicativity force
\[
  \Phi\Bigl(\sum_i a_ix^i\Bigr)
  =\sum_i\Phi(a_ix^i)
  =\sum_i\Phi(a_i)\Phi(x)^i
  =\sum_i\varphi(a_i)\alpha^i.
\]
This proves uniqueness. For existence, define $\Phi$ by this formula.
For a constant polynomial $r$, it gives $\Phi(r)=\varphi(r)$, so in
particular $\Phi(1)=1$. For $x$ it gives $\Phi(x)=\alpha$.
Since polynomials are added coefficientwise and $\varphi$ is additive,
$\Phi(f+g)=\Phi(f)+\Phi(g)$.

For multiplicativity, let $f=\sum_i a_ix^i$ and $g=\sum_j b_jx^j$.
Use the coefficient formula for $fg$ and the distributive law in $S$.
\begin{align*}
  \Phi(fg)
  &=\Phi\Bigl(\sum_k\Bigl(\sum_{i+j=k}a_ib_j\Bigr)x^k\Bigr)
   =\sum_k\Bigl(\sum_{i+j=k}\varphi(a_ib_j)\Bigr)\alpha^k\\
  &=\sum_k\Bigl(\sum_{i+j=k}\varphi(a_i)\varphi(b_j)\Bigr)\alpha^k
   =\Bigl(\sum_i\varphi(a_i)\alpha^i\Bigr)
    \Bigl(\sum_j\varphi(b_j)\alpha^j\Bigr)
   =\Phi(f)\,\Phi(g).\qedhere
\end{align*}
\end{proof}

\begin{example}[{Evaluation at $\sqrt[3]{2}$}]\label{ex:cube-root-evaluation}
Take $\varphi:\Q\to\C$ to be the inclusion and let
$\gamma=\sqrt[3]{2}$ be the real cube root of $2$. The substitution
principle gives the evaluation homomorphism $\Phi:\Q[x]\to\C$ with
$\Phi(f)=f(\gamma)$. In particular,
\[
  \Phi(x^3-2)=\gamma^3-2=0,\qquad \Phi(x-2)=\gamma-2.
\]
\end{example}

\section{Kernels of ring homomorphisms and ideals}

\begin{define}[Kernel]
Let $\varphi:R\to S$ be a ring homomorphism. Its \emph{kernel} is
\[
  \ker(\varphi)=\{x\in R:\varphi(x)=0\}.
\]
\end{define}

\begin{proposition}\label{prop:injective}
$\ker(\varphi)=0$ if and only if $\varphi$ is injective.
\end{proposition}
\begin{proof}
In particular, $\varphi$ is a homomorphism of the additive groups
underlying $R$ and $S$, and $\ker(\varphi)$ is its kernel as a group
homomorphism. The statement therefore follows from what we know about
kernels of group homomorphisms.
\end{proof}

The kernel is an additive subgroup of $R$. It is also closed under
multiplication by arbitrary elements of $R$, the defining property
of an ideal.

\begin{define}[Ideal]
A subset $I\subseteq R$ is an \emph{ideal} if
\begin{enumerate}[label=(\arabic*)]
  \item $0\in I$;
  \item if $a,b\in I$, then $a+b\in I$;
  \item if $a\in I$ and $b\in R$, then $ab\in I$.
\end{enumerate}
\end{define}

Conditions (1) and (2), together with (3) applied to $b=-1$, give
that every ideal is an additive subgroup of $R$.

\begin{lemma}
The kernel of a ring homomorphism $\varphi:R\to S$ is an ideal of $R$.
\end{lemma}
\begin{proof}
The kernel is an additive subgroup, so it contains $0$ and is
closed under addition. If $a\in\ker(\varphi)$ and $b\in R$, then
\[
  \varphi(ab)=\varphi(a)\varphi(b)=0\cdot\varphi(b)=0,
\]
so $ab\in\ker(\varphi)$.
\end{proof}

\begin{example}[Ideals of $\Z$]
For $n\geq0$, the sets
\[
  n\Z=\{0,\pm n,\pm2n,\dots\}
\]
are ideals of $\Z$. For $n\geq2$, the ideal $n\Z$ is the kernel of
reduction modulo $n$, and $1\notin n\Z$. Every ideal of $\Z$ is
an additive subgroup of $\Z$, so these are all the possibilities.
\end{example}

\section{Ideals generated by elements}

\begin{definition}[Ideal generated by elements]
If $f_1,\dots,f_n\in R$, we write $I=(f_1,\dots,f_n)$ for the
\emph{ideal generated by $f_1,\dots,f_n$}, that is, the smallest
ideal of $R$ containing $f_1,\dots,f_n$.
\end{definition}

What is it? It is the set of all combinations
\[
  (f_1,\dots,f_n)=\{a_1f_1+\cdots+a_nf_n:a_i\in R\}.
\]
Any ideal containing $f_1,\dots,f_n$ contains all such combinations
by conditions (2) and (3), so it remains to check that this set $I$
is an ideal. Clearly $0\in I$, and $I$ is closed under addition.
Moreover, if $b\in R$, then
\[
  b(a_1f_1+\cdots+a_nf_n)=(ba_1)f_1+\cdots+(ba_n)f_n\in I.
\]

\begin{example}
\begin{enumerate}[label=(\alph*)]
  \item $(1)=R$.
  \item For $a\in R$, we have $(a)=\{ba:b\in R\}=Ra$.
\end{enumerate}
\end{example}

\begin{lemma}[Ideals of a field]
A ring $R$ is a field if and only if it has exactly two distinct ideals,
$(0)$ and $(1)$.
\end{lemma}
\begin{proof}
Suppose $R$ is a field and $I\neq(0)$ is an ideal of $R$. Choose a
nonzero $a\in I$. Since $a$ is invertible, $1=a^{-1}a\in I$. Thus
$I=(1)=R$.

Conversely, suppose $(0)$ and $(1)$ are the only ideals of $R$ and
are distinct, so $1\neq0$. For any nonzero $a\in R$, the ideal $(a)$
contains $a$, so $(a)=(1)$. It follows that $1\in(a)$, so $ba=1$ for
some $b\in R$. Thus every nonzero element is invertible and $R$ is
a field.
\end{proof}

\section{Every ideal of \texorpdfstring{$F[x]$}{F[x]} is principal}

Let $F$ be a field (so, for instance, $F=\Q$, $\R$, $\C$,
$\Z/p\Z$, \dots), and let $R=F[x]$ be the ring of polynomials with
coefficients in $F$. What are the ideals of $R$? One example is
$(f(x))$ for any $f\in F[x]$.

\begin{define}[Principal ideal]
If an ideal has the form $I=(f_1,\dots,f_n)$, we say that $I$ is
\emph{generated by} $f_1,\dots,f_n$. If $I=(f)$ for a single element
$f$, we say that $I$ is \emph{principal}.
\end{define}

\begin{theorem}\label{thm:Fx-principal}
Every ideal in $F[x]$ is principal.
\end{theorem}

This is another instance of the analogy $F[x]\longleftrightarrow\Z$.

\begin{proof}
Let $I\subseteq F[x]$ be any ideal. If $I=(0)$, it is principal.
If $I\neq(0)$, then $I$ contains a nonzero polynomial of smallest
degree; call it $f\in I$, say
\[
  f(x)=a_nx^n+a_{n-1}x^{n-1}+\cdots+a_0,\qquad a_n\neq0.
\]
Since $F$ is a field and $f\in I$, so is $\frac1{a_n}f\in I$.
That is, there exists a monic polynomial
\[
  x^n+b_{n-1}x^{n-1}+\cdots+b_0\in I
\]
of smallest degree; replace $f$ by it.

Now let $g(x)\in I$. By the division algorithm for $F[x]$, since $f$
is monic, we may write
\[
  g(x)=h(x)f(x)+r(x)
\]
for some $r(x)$ which is either $0$ or of degree less than
$\deg(f)$. Since $g\in I$ and $hf\in I$,
\[
  g-hf=r\in I.
\]
If $r\neq0$, then $\deg(r)<\deg(f)$, contradicting the choice of $f$
as an element of smallest degree in $I$. Therefore $r=0$ and
$g=h\cdot f$, i.e.\ $g\in(f)$. Hence $I=(f)$.
\end{proof}

\begin{warningbox}
If $R$ is a general ring (not a field), then $R[x]$ need not be a
principal ideal domain; that is, not every ideal of $R[x]$ need be
principal.
\end{warningbox}

\begin{example}[A non-principal ideal of {$\Z[x]$}]
In $\Z[x]$, consider $I=(3,5x)$. Can we generate $I$ with just one
element? Suppose $I=(g)$ with $g\in\Z[x]$. Since $3\in(g)$, we have
$3=hg$ for some $h$, so $g$ must have degree $0$, i.e.\ $g$ is a
constant. Also $g\mid3$ and $g\mid5x$. The constant divisors of $3$
are $\pm1,\pm3$, and $\pm3$ does not divide $5x$. So $g=\pm1$ and
$I=\Z[x]$. But then
\[
  1=a(x)\cdot3+b(x)\cdot5x
\]
for some $a,b\in\Z[x]$, which is impossible: comparing constant
terms gives $1=3a(0)$ in $\Z$.
% NOTE: the notes say "impossible for degree reasons"; the argument
% given here compares constant terms instead.
Hence $I$ is not principal.
\end{example}
