\section{Review: subgroups of the integers and greatest common divisors}

Last time we classified all subgroups of $\Z$ using the following result.

\begin{theorem}
Let $H\leq(\Z,+)$ be a subgroup. Then $H=\Z d$, the set of all multiples
of $d$, for a unique integer $d\geq0$. The case $d=0$ gives the trivial
subgroup $\{0\}$.
\end{theorem}

We also stated the following result:

\begin{lemma}[Divisibility and inclusion]\label{lem:divisibility}
	For integers $r,s$,
	\be
	r\mid s\quad\Longleftrightarrow\quad\Z s\subseteq\Z r.
	\ee
\end{lemma}

\begin{proof}
	If $s=kr$ for some $k\in\Z$, then every multiple of $s$ is a multiple
	of $r$. Conversely, if $\Z s\subseteq\Z r$, then $s\in\Z r$, so $r\mid s$.
\end{proof}

\begin{example}
	The direction of inclusion reverses the order in the divisibility relation:
	\be
	2\mid6,\qquad 6\Z\subseteq2\Z.
	\ee
	Indeed, every multiple of $6$ is a multiple of $2$.
\end{example}

Recall that $\Z a+\Z b=\{ma+nb:m,n\in\Z\}$ is a subgroup of $\Z$:
it contains $0$ and is closed under sums and additive inverses.
If $a$ and $b$ are not both zero, this subgroup is nonzero, so the preceding
classification gives $\Z a+\Z b=\Z d$ for a unique $d>0$.

\begin{proposition}
	Let $a,b\in\Z$ be not both zero, and let $d>0$ satisfy
	$\Z a+\Z b=\Z d$. Then $d=\gcd(a,b)$. In particular,
	\be
	\Z a+\Z b=\Z\gcd(a,b).
	\ee
\end{proposition}

\begin{proof}
	We check two properties:
	\begin{enumerate}[label=\textbf{(\arabic*)}]
		\item $d$ divides both $a$ and $b$.
		\item Every common divisor of $a$ and $b$ divides $d$.
	\end{enumerate}
	For the first property, the inclusions
	\be
	\Z a\subseteq\Z a+\Z b=\Z d,
	\qquad
	\Z b\subseteq\Z a+\Z b=\Z d
	\ee
	imply $d\mid a$ and $d\mid b$ by Lemma~\ref{lem:divisibility}.

	For the second property, suppose $\ell\mid a$ and $\ell\mid b$.
	Then $\Z a\subseteq\Z\ell$ and $\Z b\subseteq\Z\ell$. Adding elements
	from these two subgroups gives
	\be
	\Z d=\Z a+\Z b\subseteq\Z\ell.
	\ee
	By Lemma~\ref{lem:divisibility}, $\ell\mid d$.
	Every positive common divisor $\ell$ therefore satisfies $\ell\leq d$.
	Since $d>0$ is itself a common divisor, it is the greatest common divisor.
\end{proof}

This gives B\'ezout's identity.

\begin{corollary}[B\'ezout's identity]
	Let $a,b\in\Z$ be not both zero. There exist integers $m,n$ such that
	\be
	\gcd(a,b)=ma+nb.
	\ee
\end{corollary}

\begin{proof}
	Set $d=\gcd(a,b)$. Since $\Z a+\Z b=\Z d$ and $d\in\Z d$, we have
	$d\in\Z a+\Z b$. By the definition of this subgroup, $d=ma+nb$ for some
	$m,n\in\Z$.
\end{proof}

\section{Group homomorphisms}

\begin{define}[Homomorphism]
	Let $G$ and $H$ be groups. A map $\varphi:G\to H$ is a \emph{group
		homomorphism} if
	\begin{equation}
		\varphi(ab)=\varphi(a)\varphi(b)
		\qquad\text{for every }a,b\in G.
	\end{equation}
	The products on the two sides are taken in $G$ and $H$, respectively.

	An \emph{isomorphism} is a bijective homomorphism.
\end{define}

From the product rule, we may compute:
\[
\varphi(e_G)=\varphi(e_Ge_G)=\varphi(e_G)^2,
\]
and cancellation in $H$ gives $\varphi(e_G)=e_H$. Similarly,
\[
e_H=\varphi(e_G)=\varphi(aa^{-1})
=\varphi(a)\varphi(a^{-1}),
\]
so
\begin{equation}
	\varphi(a^{-1})=\varphi(a)^{-1}.
\end{equation}

\begin{define}[Kernel and image]
	Let $\varphi:G\to H$ be a homomorphism. Its \emph{kernel} and \emph{image}
	are
	\begin{align}
		\ker(\varphi)&=\{x\in G:\varphi(x)=e_H\},\\
		\im(\varphi)&=\{y\in H:y=\varphi(x)\text{ for some }x\in G\}.
	\end{align}
\end{define}

\begin{example}[Reduction modulo $n$]
Let $n\geq1$ be an integer. Write $\overline a$ for the congruence class
of $a$ modulo $n$. The group $\Z/n\Z$ consists of these classes, with
addition $\overline a+\overline b=\overline{a+b}$.
The natural projection
\begin{equation}
  \pi_n:\Z\longrightarrow\Z/n\Z,\qquad a\longmapsto\overline a,
\end{equation}
is a homomorphism of additive groups, since
\[
  \pi_n(a+b)=\overline{a+b}=\overline a+\overline b
  =\pi_n(a)+\pi_n(b).
\]
Every congruence class has an integer representative, so $\pi_n$ is
surjective. Its kernel is
\[
  \ker(\pi_n)=\{a\in\Z:\overline a=\overline0\}
  =\{a\in\Z:n\mid a\}=n\Z.
\]
Thus $\im(\pi_n)=\Z/n\Z$, while $\pi_n$ is not injective:
the distinct integers $0$ and $n$ both map to $\overline0$.
\end{example}

\begin{proposition}
	The kernel $\ker(\varphi)$ and image $\im(\varphi)$ are subgroups of
	$G$ and $H$, respectively.
\end{proposition}
\begin{proof}
	The kernel contains $e_G$. If $a,b\in\ker(\varphi)$, then
	$\varphi(ab)=e_He_H=e_H$, and
	$\varphi(a^{-1})=\varphi(a)^{-1}=e_H$. Thus the kernel is a subgroup.
	The image contains $e_H=\varphi(e_G)$; it is closed under products because
	$\varphi(a)\varphi(b)=\varphi(ab)$ and under inverses because
	$\varphi(a)^{-1}=\varphi(a^{-1})$.
\end{proof}

\begin{lemma}[Fibres of a homomorphism]
	Let $\varphi:G\to H$ be a homomorphism. For $a,b\in G$,
	\begin{equation}
		\varphi(a)=\varphi(b)
		\quad\Longleftrightarrow\quad
		a^{-1}b\in\ker(\varphi)
		\quad\Longleftrightarrow\quad
		b^{-1}a\in\ker(\varphi).
	\end{equation}
	In particular,
	\begin{equation}
		\varphi\text{ is injective}
		\quad\Longleftrightarrow\quad
		\ker(\varphi)=\{e_G\}.
	\end{equation}
\end{lemma}

\begin{proof}
	If $\varphi(a)=\varphi(b)$, then
	\[
	\varphi(a^{-1}b)=\varphi(a)^{-1}\varphi(b)=e_H.
	\]
	The converse follows by reversing this calculation. The equivalent statement
	with $b^{-1}a$ follows by taking inverses. If $\varphi$ is injective,
	then $\varphi(a)=e_H=\varphi(e_G)$ forces $a=e_G$, so its kernel is
	trivial. Conversely, if the kernel is trivial and $\varphi(a)=\varphi(b)$,
	the first equivalence gives $a^{-1}b=e_G$, hence $a=b$.
\end{proof}

\begin{corollary}
	A homomorphism $\varphi:G\to H$ is an isomorphism if and only if
	\begin{equation}
		\ker(\varphi)=\{e_G\}
		\qquad\text{and}\qquad
		\im(\varphi)=H.
	\end{equation}
\end{corollary}

We have defined an isomorphism as a bijective homomorphism. I prefer to think
of it as a homomorphism with a homomorphic inverse: a homomorphism
$\varphi:G\to H$ together with a homomorphism
$\psi:H\to G$ satisfying
\begin{equation}
	\varphi\circ\psi=\id_H,
	\qquad
	\psi\circ\varphi=\id_G.
\end{equation}

The following proposition shows that these two descriptions are equivalent.

\begin{proposition}
	If a homomorphism $\varphi:G\to H$ is bijective, then its set-theoretic
	inverse $\psi:H\to G$ is also a homomorphism.
\end{proposition}

\begin{proof}
	Given $h_1,h_2\in H$, choose $g_1,g_2\in G$ with
	$\varphi(g_i)=h_i$. Then
	\[
	\psi(h_1h_2)=\psi\bigl(\varphi(g_1)\varphi(g_2)\bigr)
	=\psi\bigl(\varphi(g_1g_2)\bigr)=g_1g_2
	=\psi(h_1)\psi(h_2).
	\]
\end{proof}

\begin{example}[Homomorphisms determined by generators and relations]
Consider the six-element group from Section~1:
\[
  G=\{1,x,x^2,y,xy,x^2y\},
  \qquad x^3=1,\quad y^2=1,\quad yx=x^2y,
\]
and let
\[
  H=S_3=\{(1),(1\ 2),(1\ 3),(2\ 3),(1\ 2\ 3),(1\ 3\ 2)\}.
\]
Here $(1)$ denotes the identity permutation, and permutations are composed
from right to left. What maps $\varphi:G\to S_3$ are homomorphisms?

\smallskip
\noindent\textbf{Step 1: choose the images of the generators.}
Put $\sigma=\varphi(x)$ and $\tau=\varphi(y)$. If $\varphi$ is a
homomorphism, all its values are forced by these two choices:
\[
\begin{array}{c|cccccc}
  g&1&x&x^2&y&xy&x^2y\\ \hline
  \varphi(g)&(1)&\sigma&\sigma^2&\tau&\sigma\tau&\sigma^2\tau
\end{array}
\]
In particular, we cannot choose the images of the six elements independently.

\smallskip
\noindent\textbf{Step 2: preserve the relations.}
Applying $\varphi$ to each defining relation gives
\[
  \sigma^3=(1),\qquad \tau^2=(1),\qquad
  \tau\sigma=\sigma^2\tau.
\]
Thus $\sigma$ is either $(1)$ or a $3$-cycle, while $\tau$ is either
$(1)$ or a transposition. The order of an image may be smaller than the
order of the original element: for example, $\varphi(x)=(1)$ is allowed.
The last relation imposes an additional compatibility condition.

These conditions are also sufficient. Indeed, write each element of $G$
uniquely as $x^iy^j$, where $0\leq i<3$ and $0\leq j<2$. Since
$yx=x^{-1}y$, multiplication in $G$ is given by
\[
  (x^iy^j)(x^ky^\ell)=x^{\,i+(-1)^jk}y^{\,j+\ell},
\]
with the exponents reduced modulo $3$ and $2$, respectively. If
$\sigma,\tau$ satisfy the three displayed relations, then
$\tau\sigma=\sigma^{-1}\tau$, so their products obey the same rule:
\[
  (\sigma^i\tau^j)(\sigma^k\tau^\ell)
  =\sigma^{\,i+(-1)^jk}\tau^{\,j+\ell}.
\]
Consequently, $\varphi(x^iy^j)=\sigma^i\tau^j$ defines a homomorphism.

\smallskip
\noindent\textbf{Step 3: list the possibilities.}
\begin{enumerate}[label=\textbf{(\arabic*)}]
  \item If $\sigma=(1)$, the compatibility condition is automatic.
  We may take $\tau=(1)$ or any of the three transpositions.
  The first choice gives the trivial homomorphism, with kernel $G$ and
  image $\{(1)\}$. For each transposition $\tau$, we obtain a homomorphism
  with
  \[
    \ker(\varphi)=\{1,x,x^2\},\qquad
    \im(\varphi)=\{(1),\tau\}.
  \]
  These are four homomorphisms in total.

  \item If $\sigma$ is a $3$-cycle, we cannot take $\tau=(1)$:
  the last relation would give $\sigma=\sigma^2$, hence $\sigma=(1)$.
  Every transposition does work, because conjugating a $3$-cycle by a
  transposition reverses it:
  \[
    \tau\sigma\tau^{-1}=\sigma^{-1}=\sigma^2.
  \]
  There are two choices for $\sigma$ and three for $\tau$, giving six
  homomorphisms. Each is an isomorphism: the three elements
  $(1),\sigma,\sigma^2$ are distinct, as are
  $\tau,\sigma\tau,\sigma^2\tau$, and these two sets are disjoint.
  Indeed, an equality $\sigma^i\tau=\sigma^k$ would make the
  transposition $\tau$ a power of a $3$-cycle. Thus all six images are
  distinct, so $\ker(\varphi)=\{1\}$ and $\im(\varphi)=S_3$.
\end{enumerate}
For instance, choosing $\varphi(x)=(1\ 2\ 3)$ and
$\varphi(y)=(1\ 2)$ gives the isomorphism
\[
\begin{array}{c|cccccc}
  g&1&x&x^2&y&xy&x^2y\\ \hline
  \varphi(g)&(1)&(1\ 2\ 3)&(1\ 3\ 2)&(1\ 2)&(1\ 3)&(2\ 3)
\end{array}
\]
There are therefore exactly $4+6=10$ homomorphisms $G\to S_3$, of which
six are isomorphisms.
\end{example}
