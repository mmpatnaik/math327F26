\section{Group homomorphisms}

\begin{definition}[Homomorphism]
Let $G$ and $H$ be groups. A map $\varphi:G\to H$ is a \emph{group
homomorphism} if
\begin{equation}
  \varphi(ab)=\varphi(a)\varphi(b)
  \qquad\text{for every }a,b\in G.
\end{equation}
The products on the two sides are taken in $G$ and $H$, respectively.
\end{definition}

Although one sometimes includes preservation of the identity in the
definition, it follows from the product rule. Indeed,
\[
  \varphi(e_G)=\varphi(e_Ge_G)=\varphi(e_G)^2,
\]
and cancellation in $H$ gives $\varphi(e_G)=e_H$. Similarly,
\[
  e_H=\varphi(e_G)=\varphi(aa^{-1})
  =\varphi(a)\varphi(a^{-1}),
\]
so
\begin{equation}
  \varphi(a^{-1})=\varphi(a)^{-1}.
\end{equation}

Here is the same argument in the order used in class. First apply the
product rule to $a\cdot 1=a$:
\[
  \varphi(a)=\varphi(a\cdot1)=\varphi(a)\varphi(1).
\]
Cancelling $\varphi(a)$ gives $\varphi(1)=1$. Then apply the product rule
to $aa^{-1}=1$:
\[
  1=\varphi(1)=\varphi(aa^{-1})
   =\varphi(a)\varphi(a^{-1}),
\]
which says exactly that $\varphi(a^{-1})$ is the inverse of $\varphi(a)$.

\begin{remark}
Thus it is equivalent to define a homomorphism as a map satisfying only
$\varphi(ab)=\varphi(a)\varphi(b)$. Preservation of identities and inverses
then comes for free.
\end{remark}

\section{Kernel, image, and fibres}

\begin{definition}
Let $\varphi:G\to H$ be a homomorphism. Its \emph{kernel} and \emph{image}
are
\begin{align}
  \ker(\varphi)&=\{x\in G:\varphi(x)=e_H\},\\
  \im(\varphi)&=\{y\in H:y=\varphi(x)\text{ for some }x\in G\}.
\end{align}
\end{definition}

\begin{proposition}
The sets $\ker(\varphi)\leq G$ and $\im(\varphi)\leq H$ are subgroups.
\end{proposition}

\begin{proof}
The kernel contains $e_G$. If $a,b\in\ker(\varphi)$, then
$\varphi(ab)=e_He_H=e_H$, and
$\varphi(a^{-1})=\varphi(a)^{-1}=e_H$. Thus the kernel is a subgroup.
The image contains $e_H=\varphi(e_G)$; it is closed under products because
$\varphi(a)\varphi(b)=\varphi(ab)$ and under inverses because
$\varphi(a)^{-1}=\varphi(a^{-1})$.
\end{proof}

\begin{lemma}[Fibres of a homomorphism]
For $a,b\in G$,
\begin{equation}
  \varphi(a)=\varphi(b)
  \quad\Longleftrightarrow\quad
  a^{-1}b\in\ker(\varphi)
  \quad\Longleftrightarrow\quad
  b^{-1}a\in\ker(\varphi).
\end{equation}
In particular,
\begin{equation}
  \varphi\text{ is injective}
  \quad\Longleftrightarrow\quad
  \ker(\varphi)=\{e_G\}.
\end{equation}
\end{lemma}

\begin{proof}
If $\varphi(a)=\varphi(b)$, then
\[
  \varphi(a^{-1}b)=\varphi(a)^{-1}\varphi(b)=e_H.
\]
The converse follows by reversing this calculation. The equivalent statement
with $b^{-1}a$ follows by taking inverses. For the last assertion, apply the
first statement to an equality $\varphi(a)=\varphi(b)$.
\end{proof}

More explicitly, the two directions of the first equivalence are
\begin{align*}
  \varphi(a)=\varphi(b)
  &\Longrightarrow
  \varphi(a^{-1}b)=\varphi(a)^{-1}\varphi(b)=e_H,\\
  a^{-1}b\in\ker(\varphi)
  &\Longrightarrow
  e_H=\varphi(a^{-1}b)=\varphi(a)^{-1}\varphi(b)
  \Longrightarrow \varphi(a)=\varphi(b).
\end{align*}
Thus the elements in any one fibre of $\varphi$ differ from one another by
elements of the kernel. This observation will later become the language of
cosets.

\begin{corollary}
A homomorphism $\varphi:G\to H$ is an isomorphism if and only if
\begin{equation}
  \ker(\varphi)=\{e_G\}
  \qquad\text{and}\qquad
  \im(\varphi)=H.
\end{equation}
\end{corollary}

Recall that an isomorphism is a homomorphism with a homomorphic inverse.
The preceding corollary uses the following useful fact.

In other words, an isomorphism $\varphi:G\to H$ comes with a homomorphism
$\psi:H\to G$ satisfying
\begin{equation}
  \varphi\circ\psi=\id_H,
  \qquad
  \psi\circ\varphi=\id_G.
\end{equation}

\begin{proposition}
If a homomorphism $\varphi:G\to H$ is bijective, then its set-theoretic
inverse $\psi:H\to G$ is also a homomorphism.
\end{proposition}

\begin{proof}
Given $h_1,h_2\in H$, choose $g_1,g_2\in G$ with
$\varphi(g_i)=h_i$. Then
\[
  \psi(h_1h_2)=\psi\bigl(\varphi(g_1)\varphi(g_2)\bigr)
  =\psi\bigl(\varphi(g_1g_2)\bigr)=g_1g_2
  =\psi(h_1)\psi(h_2).
\]
\end{proof}

\section{Determinant and sign}

\begin{example}[The determinant homomorphism]
Multiplicativity of the determinant says that
\begin{equation}
  \det:\GL_n(\R)\longrightarrow\R^*,
  \qquad A\longmapsto\det(A),
\end{equation}
is a homomorphism. Its kernel is
\[
  \ker(\det)=\SL_n(\R)
  =\{A\in\GL_n(\R):\det(A)=1\}.
\]
\end{example}

There are two natural ways to try to define the sign of a permutation.

\subsection*{Approach A: count transpositions}

Write $\sigma\in S_n$ as a product of transpositions.  One would like to set
\[
  \sgn(\sigma)=
  \begin{cases}
    +1,&\text{if an even number of transpositions is used},\\
    -1,&\text{if an odd number of transpositions is used}.
  \end{cases}
\]
The multiplication rule in $\{\pm1\}$ has exactly the desired parity table:
\[
\begin{array}{c|cc}
 &\text{even}&\text{odd}\\ \hline
\text{even}&\text{even}&\text{odd}\\
\text{odd}&\text{odd}&\text{even}
\end{array}
\]
The difficulty is that a permutation has many decompositions into
transpositions.  We must first prove that the parity of the number of factors
does not depend on the chosen decomposition.  Permutation matrices provide a
clean way to do this.

\subsection*{Approach B: permutation matrices}

We next construct the sign homomorphism geometrically. For
$\sigma\in S_n$, let $e_1,\dots,e_n$ be the standard basis of $\R^n$ and
define the permutation matrix $[\sigma]$ by
\begin{equation}
  [\sigma]e_i=e_{\sigma(i)}.
\end{equation}
Equivalently, the $i$th column of $[\sigma]$ is $e_{\sigma(i)}$.
If $E_{ij}$ denotes the matrix with a $1$ in row $i$, column $j$, and zeros
elsewhere, then
\begin{equation}
  [\sigma]=\sum_{i=1}^n E_{\sigma(i),i}.
\end{equation}

\begin{example}
For $\sigma=(1\ 2\ 3)\in S_3$,
\[
  [\sigma]=
  \begin{pmatrix}
    0&0&1\\
    1&0&0\\
    0&1&0
  \end{pmatrix},
  \qquad
  [\sigma]
  \begin{pmatrix}x_1\\x_2\\x_3\end{pmatrix}
  =
  \begin{pmatrix}x_3\\x_1\\x_2\end{pmatrix}.
\]
More generally,
\[
  [\sigma](x_1,\dots,x_n)^T
  =(x_{\sigma^{-1}(1)},\dots,x_{\sigma^{-1}(n)})^T.
\]
\end{example}

\begin{warningbox}
The entries of a vector are rearranged using $\sigma^{-1}$, not $\sigma$.
The reason is that the $i$th basis vector is sent to $e_{\sigma(i)}$:
the entry originally in position $i$ therefore appears in position
$\sigma(i)$.
\end{warningbox}

The assignment $\sigma\mapsto[\sigma]$ is a homomorphism
$S_n\to\GL_n(\R)$. Since a permutation matrix has integer entries, its
determinant is an integer. It is also invertible, and
\[
  \det([\sigma])\det([\sigma^{-1}])=1.
\]
Hence $\det([\sigma])\in\{+1,-1\}$.

\begin{definition}[Sign]
The \emph{sign homomorphism} is
\begin{equation}
  \sgn:S_n\longrightarrow\{\pm1\},
  \qquad \sgn(\sigma)=\det([\sigma]).
\end{equation}
A permutation is \emph{even} if its sign is $+1$, and \emph{odd} if its
sign is $-1$.
\end{definition}

The map is a homomorphism because
\[
  \sgn(\sigma\tau)=\det([\sigma\tau])
  =\det([\sigma][\tau])=\sgn(\sigma)\sgn(\tau).
\]

A transposition exchanges two columns of the identity matrix and hence has
determinant $-1$.  Therefore, if
$\sigma=\tau_1\cdots\tau_r$ is any decomposition into transpositions, then
\begin{equation}
  \sgn(\sigma)=(-1)^r.
\end{equation}
This proves that every decomposition of a fixed permutation has the same
parity and justifies Approach A.

\section{The centre of a group}

\begin{definition}
The \emph{centre} of a group $G$ is
\begin{equation}
  Z(G)=\{z\in G:zx=xz\text{ for every }x\in G\}.
\end{equation}
\end{definition}

\begin{proposition}
The centre $Z(G)$ is a subgroup of $G$.
\end{proposition}

\begin{proof}
The identity commutes with every element. If $z_1,z_2\in Z(G)$ and $x\in G$,
then
\[
  z_1z_2x=z_1xz_2=xz_1z_2.
\]
Also $zx=xz$ implies $z^{-1}x=xz^{-1}$, so $z^{-1}\in Z(G)$.
\end{proof}

For the inverse calculation, start with $zx=xz$ and multiply on the left and
right by $z^{-1}$:
\[
  xz^{-1}=z^{-1}x.
\]
Thus all three subgroup conditions are visible directly.

\begin{example}[The centre of $\SL_2(\R)$]
Let
\[
  z=\begin{pmatrix}a&b\\c&d\end{pmatrix}\in Z(\SL_2(\R)).
\]
Commuting with
$J=\begin{pmatrix}0&1\\-1&0\end{pmatrix}$ gives $a=d$ and $b=-c$.
Commuting with
$U=\begin{pmatrix}1&1\\0&1\end{pmatrix}$ then gives $b=0$.
Therefore $z=aI_2$, and the condition $\det z=1$ gives $a^2=1$.
Consequently,
\begin{equation}
  Z(\SL_2(\R))=\{I_2,-I_2\}.
\end{equation}
\end{example}

Indeed, the first commutation relation is
\[
  \begin{pmatrix}a&b\\c&d\end{pmatrix}
  \begin{pmatrix}0&1\\-1&0\end{pmatrix}
  =
  \begin{pmatrix}0&1\\-1&0\end{pmatrix}
  \begin{pmatrix}a&b\\c&d\end{pmatrix},
\]
or
\[
  \begin{pmatrix}-b&a\\-d&c\end{pmatrix}
  =
  \begin{pmatrix}c&d\\-a&-b\end{pmatrix}.
\]
Hence $z=\begin{pmatrix}a&b\\-b&a\end{pmatrix}$.  The second relation becomes
\[
  \begin{pmatrix}a&a+b\\-b&a-b\end{pmatrix}
  =
  \begin{pmatrix}a-b&a+b\\-b&a\end{pmatrix},
\]
so $b=0$, as claimed.

\section{Isomorphisms and automorphisms}

We begin by recalling the definitions from the preceding lecture.  A map
$\varphi:G\to H$ is a homomorphism when
\[
  \varphi(ab)=\varphi(a)\varphi(b)
  \qquad(a,b\in G).
\]
It then follows that $\varphi(e_G)=e_H$ and
$\varphi(x^{-1})=\varphi(x)^{-1}$.

\begin{definition}
For groups $G$ and $H$, write
\[
  \Hom(G,H)=\{\varphi:G\to H:\varphi\text{ is a homomorphism}\}.
\]
An \emph{isomorphism} $\varphi:G\to H$ is a homomorphism admitting a
homomorphic inverse $\psi:H\to G$. We write $G\cong H$ if an isomorphism
exists, and $\Iso(G,H)$ for the set of all such isomorphisms.
\end{definition}

\begin{proposition}
A homomorphism is an isomorphism if and only if it is bijective. Equivalently,
\[
  \varphi:G\to H\text{ is an isomorphism}
  \quad\Longleftrightarrow\quad
  \ker(\varphi)=\{e_G\}\text{ and }\im(\varphi)=H.
\]
\end{proposition}

The two conditions in the displayed formula separate bijectivity into its
two parts:
\[
  \varphi\text{ injective}\Longleftrightarrow\ker(\varphi)=\{e_G\},
  \qquad
  \varphi\text{ surjective}\Longleftrightarrow\im(\varphi)=H.
\]

\begin{definition}
An isomorphism $G\to G$ is called an \emph{automorphism} of $G$. The set of
all automorphisms is denoted by $\Aut(G)$.
\end{definition}

\section{Conjugation}

The kernel of a homomorphism is not just a subgroup: it has an additional
symmetry. To describe it, fix $g\in G$.

\begin{definition}
The map
\begin{equation}
  c_g:G\longrightarrow G,
  \qquad c_g(x)=gxg^{-1},
\end{equation}
is called \emph{conjugation by $g$}.
\end{definition}

\begin{proposition}
For every $g\in G$, the map $c_g$ is an automorphism of $G$, with inverse
$c_{g^{-1}}$.
\end{proposition}

\begin{proof}
For $a,b\in G$,
\[
  c_g(ab)=gabg^{-1}=(gag^{-1})(gbg^{-1})=c_g(a)c_g(b).
\]
Moreover,
\[
  c_gc_{g^{-1}}(x)=g(g^{-1}xg)g^{-1}=x,
  \qquad
  c_{g^{-1}}c_g(x)=x.
\]
\end{proof}

\begin{remark}
Conjugation distributes over products:
\[
  g(a_1\cdots a_r)g^{-1}
  =(ga_1g^{-1})\cdots(ga_rg^{-1}).
\]
Consequently, isomorphisms and automorphisms preserve every property that can
be expressed purely in terms of the group operation.
\end{remark}

For two factors, the inserted identity $g^{-1}g=e$ makes the calculation
transparent:
\[
  g(ab)g^{-1}=ga(g^{-1}g)bg^{-1}
  =(gag^{-1})(gbg^{-1}).
\]

For permutations, conjugation has a particularly useful description. If
$\tau=(i_1\ \cdots\ i_k)$ and $\sigma\in S_n$, then
\begin{equation}
  \sigma\tau\sigma^{-1}
  =\bigl(\sigma(i_1)\ \cdots\ \sigma(i_k)\bigr).
\end{equation}
Thus conjugation preserves cycle type.

To see the displayed formula directly, follow an entry of the cycle:
\[
  \sigma(i_1)\longmapsto\sigma(i_2)
  \longmapsto\cdots\longmapsto\sigma(i_k)
  \longmapsto\sigma(i_1).
\]

\section{Normal subgroups}

\begin{definition}
A subgroup $N\leq G$ is \emph{normal} if
\begin{equation}
  gNg^{-1}\subseteq N
  \qquad\text{for every }g\in G.
\end{equation}
We write $N\trianglelefteq G$.
\end{definition}

\begin{remark}
The inclusion in the definition is automatically an equality. Applying it
to $g^{-1}$ gives $g^{-1}Ng\subseteq N$; conjugating this inclusion by $g$
gives $N\subseteq gNg^{-1}$. Hence
\begin{equation}
  gNg^{-1}=N\qquad(g\in G).
\end{equation}
\end{remark}

Normal subgroups are useful for two related reasons.  First, they are the
subgroups that can occur as kernels of homomorphisms.  Second, they are the
subgroups for which the cosets can be made into a group.  This second point
will lead to quotient groups.

\begin{proposition}
If $\varphi:G\to H$ is a homomorphism, then
$\ker(\varphi)\trianglelefteq G$.
\end{proposition}

\begin{proof}
Let $x\in\ker(\varphi)$ and $g\in G$. Then
\[
  \varphi(gxg^{-1})
  =\varphi(g)\varphi(x)\varphi(g)^{-1}
  =\varphi(g)e_H\varphi(g)^{-1}=e_H.
\]
Thus $gxg^{-1}\in\ker(\varphi)$.
\end{proof}

\begin{example}
Every subgroup of an abelian group is normal, since $ghg^{-1}=h$.
Also, $\{e_G\}$ and $G$ are normal in every group.
\end{example}

When $N$ is neither $\{e_G\}$ nor $G$, it is a \emph{proper nontrivial normal
subgroup}.  The absence of such subgroups is important enough to name.

\begin{definition}
A group $G$ is \emph{simple} if its only normal subgroups are $\{e_G\}$ and
$G$.
\end{definition}

\begin{example}[Normal subgroups of $S_3$]
Write
\[
  S_3=\{1,x,x^2,y,xy,yx\},
  \qquad x^3=y^2=1,
  \qquad yx=x^2y.
\]
The subgroup $H=\langle x\rangle=\{1,x,x^2\}$ is normal. Conjugation by
elements of $H$ leaves $H$ fixed, and
\[
  yHy^{-1}=\{1,yxy,yx^2y\}=\{1,x^2,x\}=H.
\]
Equivalently, $H$ is the kernel of the sign homomorphism
$\sgn:S_3\to\{\pm1\}$.

In contrast, $L=\langle y\rangle=\{1,y\}$ is not normal because
\[
  xLx^{-1}=\{1,xyx^{-1}\}=\{1,yx\}\neq L.
\]
The three even permutations are $1,(1\ 2\ 3),(1\ 3\ 2)$, so
\[
  \ker(\sgn)=\{1,x,x^2\}.
\]
\end{example}

Here are the conjugation calculations from class in expanded form.  Since
$y^{-1}=y$ and $yx=x^2y$,
\begin{align*}
  yxy^{-1}&=yxy=x^2,\\
  yx^2y^{-1}&=(yxy)(yxy)=x^2x^2=x.
\end{align*}
Hence $yHy^{-1}=H$.  For $L=\{1,y\}$, however,
\[
  xLx^{-1}=\{1,xyx^{-1}\}=\{1,yx\},
\]
which is a different two-element subgroup.  Thus the two cyclic subgroups
behave differently under conjugation even though both are abelian in their
own right.

In cycle notation, the six elements of $S_3$ split as
\[
  \underbrace{\id,(1\ 2\ 3),(1\ 3\ 2)}_{\text{even}}
  \qquad\text{and}\qquad
  \underbrace{(1\ 2),(1\ 3),(2\ 3)}_{\text{odd}}.
\]
The even part is precisely $H=\ker(\sgn)$, which gives a second proof that
$H\trianglelefteq S_3$.
