\makelecturetitle

Last time we studied the notion of a left action of a group $G$ on a set $S$.
For every $s\in S$, we also defined its orbit
\begin{equation}\label{orbit}
  \mathcal O_s:=\{g\cdot s\mid g\in G\}
\end{equation}
as a subset of $S$. Moreover, we showed that the orbits partition $S$;
that is, we have a disjoint union
\[
  S=\bigsqcup_{s\in\Xi}\mathcal O_s,
\]
where $\Xi\subseteq S$ contains one representative from each orbit.
Finally, for each $s\in S$, we introduced the stabilizer subgroup
\begin{equation}\label{stab:s}
  G_s:=\{g\in G\mid g\cdot s=s\}.
\end{equation}

An important observation which I forgot to mention last time is the following:


\begin{lemma}[Stabilizers in the same orbit]\label{lem:conjugate-stabilizers}
Let $s\in S$ and $a\in G$. If $s'=a\cdot s$, then $G_{s'}=aG_sa^{-1}$.
\end{lemma}
\begin{proof}
For $x\in G$,
\begin{align*}
  x\in G_{s'}
  &\iff x\cdot(a\cdot s)=a\cdot s\\
  &\iff (a^{-1}xa)\cdot s=s\\
  &\iff a^{-1}xa\in G_s.
\end{align*}
This is equivalent to $x\in aG_sa^{-1}$.
\end{proof}

Thus stabilizers of points in the same orbit are conjugate subgroups.
A stabilizer need not be normal in $G$. 


\Needspace{12\baselineskip}
\section{The orbit--stabilizer theorem}

The orbits and stabilizers are related by the following important result.


\begin{theorem}[Orbit--stabilizer]\label{thm:orbit-stabilizer}
For any action $G\curvearrowright S$ and $s\in S$, there is a
bijection of sets
\[
  G/G_s\longrightarrow\mathcal O_s,
  \qquad gG_s\longmapsto g\cdot s.
\]
Here $G/G_s$ is the set of left cosets; $G_s$ need not be normal.
\end{theorem}
\begin{proof}
If $gG_s=\widetilde gG_s$, then
$\widetilde g^{-1}g\in G_s$ and hence
$g\cdot s=\widetilde g\cdot s$. Thus the map is well defined.
It is surjective by the definition of $\mathcal O_s$.
If $g\cdot s=\widetilde g\cdot s$, then
$\widetilde g^{-1}g\in G_s$, so $gG_s=\widetilde gG_s$.
Therefore it is injective.
\end{proof}

\begin{corollary}\label{cor:orbit-size}
If $G$ is finite, then
\[
  |\mathcal O_s|=[G:G_s],
  \qquad |G|=|\mathcal O_s|\,|G_s|.
\]
In particular, every orbit size divides $|G|$.
\end{corollary}



\section{Conjugation as an action}

Every group acts on itself by \emph{conjugation}:
\[
  g\cdot x=gxg^{-1}.
\]
Indeed, $1\cdot x=x$ and
$(gh)\cdot x=ghx(gh)^{-1}=g\cdot(h\cdot x)$.


\begin{define}[Centralizer and conjugacy class]
The stabilizer of $x\in G$ under conjugation is its
\emph{centralizer}
\[
  C_G(x)=\{g\in G:gxg^{-1}=x\}
        =\{g\in G:gx=xg\}.
\]
Its orbit is the \emph{conjugacy class}
\[
  \mathcal{O}_G(x)=\{gxg^{-1}:g\in G\}.
\]
\end{define}

Thus, for finite $G$, orbit--stabilizer gives
\[
  |\mathcal{O}_G(x)|=[G:C_G(x)].
\]

The center of $G$ is
\[
  Z(G)=\{z\in G:zx=xz\text{ for every }x\in G\}
      =\bigcap_{x\in G}C_G(x).
\]
In particular, $Z(G)\leq C_G(x)$ for every $x\in G$.

\begin{lemma}\label{lem:central-elements}
For $x\in G$, the following are equivalent:
\[
  x\in Z(G),\qquad C_G(x)=G,\qquad\mathcal{O}_G(x)=\{x\}.
\]
\end{lemma}
\begin{proof}
Each condition says that every $g\in G$ satisfies $gxg^{-1}=x$.
\end{proof}

\section{The class equation}

The orbits of any action partition the underlying set. Suppose $G$ is
finite, and choose representatives $x_1,\ldots,x_k$, one from each
conjugacy class. Then
\[
  G=\mathcal{O}_G(x_1)\sqcup\cdots\sqcup\mathcal{O}_G(x_k).
\]

\Needspace{10\baselineskip}
\begin{theorem}[Class equation]\label{thm:class-equation}
If $G$ is finite and $x_1,\ldots,x_k$ represent its distinct
conjugacy classes, then
\[
  |G|=\sum_{i=1}^k|\mathcal{O}_G(x_i)|,
  \qquad
  |\mathcal{O}_G(x_i)|=[G:C_G(x_i)].
\]
In particular, every conjugacy class size divides $|G|$.
\end{theorem}
\begin{proof}
The first equality counts the elements of the disjoint union.
The second is orbit--stabilizer for the conjugation action.
\end{proof}

The singleton classes are exactly the sets $\{z\}$ with $z\in Z(G)$.
Consequently, another useful form of the class equation is
\[
  |G|=|Z(G)|+\sum_{x_i\notin Z(G)}[G:C_G(x_i)],
\]
where the sum contains one representative from each noncentral class.

\section{Conjugacy in the symmetric group}

Recall that conjugation relabels the entries of a cycle:
\[
  \sigma(i_1\,i_2\,\cdots\,i_r)\sigma^{-1}
  =(\sigma(i_1)\,\sigma(i_2)\,\cdots\,\sigma(i_r)).
\]
More generally, two permutations in $S_n$ are conjugate if and only
if their decompositions into disjoint cycles have the same cycle
lengths, counting fixed points as cycles of length $1$. This collection
of cycle lengths, with repetitions, is called the \emph{cycle type}.

\begin{example}[The class equation of $S_3$]\label{ex:s3}
There are three cycle types in $S_3$, giving the classes
\[
  \{1\},\qquad
  \{(1\,2),(1\,3),(2\,3)\},\qquad
  \{(1\,2\,3),(1\,3\,2)\}.
\]
Thus the class equation is
\[
  6=1+3+2.
\]
In particular, $Z(S_3)=\{1\}$, since only the identity has a
singleton conjugacy class.
\end{example}

\begin{example}[Centralizers in $S_3$]
Orbit--stabilizer determines the orders of the centralizers:
\[
\begin{array}{c|c|c}
  x & |\mathcal{O}_{S_3}(x)| & C_{S_3}(x)\\ \hline
  1 & 1 & S_3\\[2pt]
  (1\,2\,3) & 2 & \langle(1\,2\,3)\rangle\\[2pt]
  (1\,3) & 3 & \{1,(1\,3)\}
\end{array}
\]
For example, the centralizer of $(1\,2\,3)$ has order $3$ and
contains its cyclic subgroup of order $3$, so the two are equal.

By Lemma~\ref{lem:conjugate-stabilizers}, conjugate elements have
conjugate centralizers:
\[
  C_G(axa^{-1})=aC_G(x)a^{-1}.
\]
The two $3$-cycles have the same centralizer, since
$\langle(1\,2\,3)\rangle$ has index $2$ in $S_3$ and is therefore normal. The three transpositions instead have three
distinct, conjugate centralizers of order $2$:
\[
  \langle(1\,2)\rangle,\qquad
  \langle(1\,3)\rangle,\qquad
  \langle(2\,3)\rangle.
\]
These, together with $\{1\}$, $A_3$, and $S_3$, are all the
subgroups of $S_3$.
\end{example}

\section{A matrix group over a finite field}

Let $\F_3=\Z/3\Z=\{\overline0,\overline1,\overline2\}$.
Matrix operations over $\F_3$ use arithmetic modulo $3$.
We consider
\[
  \SL_2(\F_3)=
  \left\{\begin{pmatrix}a&b\\c&d\end{pmatrix}:
     a,b,c,d\in\F_3,\ ad-bc=1\right\}.
\]

\begin{example}[The order of $\SL_2(\F_3)$]
First count the matrices in $\GL_2(\F_3)$ by choosing their columns.
The first column $v_1$ can be any nonzero vector in $\F_3^2$,
giving $3^2-1=8$ choices. The second column must lie outside the
span of $v_1$. That span contains $3$ vectors, so there are
$3^2-3=6$ choices. Hence
\[
  |\GL_2(\F_3)|=8\cdot6=48.
\]
The determinant homomorphism
\[
  \det:\GL_2(\F_3)\longrightarrow\F_3^\times
\]
is surjective: the diagonal matrix $\operatorname{diag}(a,1)$
has determinant $a$ for each $a\in\F_3^\times$.
Its kernel is $\SL_2(\F_3)$, and $|\F_3^\times|=2$.
The counting formula for a homomorphism therefore gives
\[
  |\SL_2(\F_3)|=\frac{48}{2}=24.
\]
\end{example}
