\section{The universal property of quotients}

Last time we let $R$ be a commutative ring and $I\subseteq R$ an
ideal. The quotient ring $R/I$ consists of the cosets $a+I$ with
$a\in R$, and the map
\[
  \pi:R\to R/I,\qquad a\mapsto a\bmod I=a+I,
\]
is a surjective ring homomorphism.

\begin{theorem}[Universal property of quotients]\label{thm:universal}
Let $\varphi:R\to S$ be a ring homomorphism and let $I$ be an ideal
of $R$ with $\varphi(I)=0$ in $S$. Then there is a unique ring
homomorphism $\tilde\varphi:R/I\to S$ making the following diagram
commute:
\[
  \begin{tikzcd}
    R \arrow[r,"\varphi"] \arrow[d,"\pi"'] & S\\
    R/I \arrow[ur,dashed,"\tilde\varphi"'] &
  \end{tikzcd}
\]
That is, $\tilde\varphi\circ\pi=\varphi$.
\end{theorem}
\begin{proof}
Commutativity forces the definition
\[
  \tilde\varphi(a+I)=\varphi(a),
\]
which gives uniqueness. This is well defined: if $a+I=a'+I$, then
$a'-a\in I$, so $\varphi(a')-\varphi(a)=\varphi(a'-a)=0$.
Hence $\varphi(a)=\varphi(a')$. Since $\varphi$ is a ring
homomorphism and the operations on $R/I$ are defined using
representatives, $\tilde\varphi$ is a ring homomorphism.
\end{proof}

\section{The first isomorphism theorem for rings}

As a consequence, we obtain the first isomorphism theorem for rings.

\begin{definition}
A ring homomorphism $\varphi:R\to S$ is an \emph{isomorphism} if it
is bijective.
\end{definition}

\begin{remark}
Equivalently:
\begin{itemize}
  \item $\varphi$ is a homomorphism with $\ker(\varphi)=0$ that is
        surjective; or
  \item $\varphi$ is a ring homomorphism that is an isomorphism of
        the underlying additive groups.
\end{itemize}
\end{remark}

\begin{theorem}[First isomorphism theorem for rings]\label{thm:first-iso}
If $\varphi:R\to S$ is a surjective homomorphism of rings, then
there is an isomorphism of rings
\[
  \bar\varphi:R/\ker(\varphi)\xrightarrow{\ \sim\ }S.
\]
\end{theorem}
\begin{proof}
Since $\varphi(\ker\varphi)=0$, Theorem~\ref{thm:universal} gives a
ring homomorphism $\bar\varphi$ fitting into the diagram
\[
  \begin{tikzcd}
    R \arrow[r,"\varphi"] \arrow[d] & S\\
    R/\ker(\varphi) \arrow[ur,dashed,"\bar\varphi"'] &
  \end{tikzcd}
\]
By the first isomorphism theorem for groups, applied to the
additive groups, $\bar\varphi$ is a bijection. A bijective ring
homomorphism is an isomorphism of rings.
\end{proof}

For any ring homomorphism $\varphi:R\to S$, applying the theorem to
$\varphi$ with codomain restricted to its image gives
\[
  R/\ker(\varphi)\cong\im(\varphi).
\]

\section{The correspondence theorem}

As in group theory, we also have a correspondence theorem in ring
theory. Recall the group-theoretic version: if $\varphi:G\to H$ is a
surjective homomorphism with kernel $K=\ker(\varphi)$, there is a
bijective correspondence
\[
  \{\text{subgroups }L\text{ of }G\text{ with }L\supseteq K\}
  \longleftrightarrow
  \{\text{subgroups }L'\text{ of }H\}.
\]

\begin{theorem}[Correspondence theorem for rings]\label{thm:correspondence}
Let $\varphi:R\to S$ be a surjective homomorphism of rings and let
$K=\ker(\varphi)$.
\begin{enumerate}
  \item There is a bijective correspondence
  \[
    \{\text{ideals }I\text{ of }R\text{ with }I\supseteq K\}
    \longleftrightarrow
    \{\text{ideals }J\text{ of }S\},
  \]
  given by $I\mapsto\varphi(I)$, with inverse $J\mapsto\varphi^{-1}(J)$.
  \item If $I\leftrightarrow J$ under this correspondence, then
  \[
    R/I\cong S/J
  \]
  as rings.
\end{enumerate}
\end{theorem}
\begin{proof}
\emph{Preimages of ideals are ideals.}
Let $\varphi:R\to S$ be any ring homomorphism and let $J$ be an
ideal of $S$. We check that $\varphi^{-1}(J)$ is an ideal of $R$.
\begin{itemize}
  \item If $a,b\in\varphi^{-1}(J)$, then $\varphi(a),\varphi(b)\in J$,
  so $\varphi(a+b)=\varphi(a)+\varphi(b)\in J$. Thus
  $a+b\in\varphi^{-1}(J)$.
  \item $\varphi(0)=0\in J$, so $0\in\varphi^{-1}(J)$.
  \item If $a\in\varphi^{-1}(J)$ and $r\in R$, then
  $\varphi(ra)=\varphi(r)\varphi(a)\in J$, because $\varphi(r)\in S$,
  $\varphi(a)\in J$, and $J$ is an ideal of $S$. Thus
  $ra\in\varphi^{-1}(J)$.
\end{itemize}
Moreover, $\varphi^{-1}(J)\supseteq\varphi^{-1}(0)=K$.

\emph{Images of ideals are ideals.}
Now assume that $\varphi$ is surjective: for every $s\in S$ there is
$r\in R$ with $\varphi(r)=s$. Let $I$ be an ideal of $R$ with
$I\supseteq K$. We check that $\varphi(I)$ is an ideal of $S$.
It is closed under addition and contains $0$. For the absorption
property, let $a\in\varphi(I)$ and $s\in S$. Then $a=\varphi(b)$ for
some $b\in I$, and by surjectivity $s=\varphi(r)$ for some $r\in R$.
Hence
\[
  s\cdot a=\varphi(r)\varphi(b)=\varphi(r b)\in\varphi(I),
\]
since $rb\in I$. The rest of the verification that $\varphi(I)$ is
an ideal of $S$ is left as an exercise.

\emph{Bijectivity.}
Ideals $I\subseteq R$ and $J\subseteq S$ are in particular additive
subgroups, and $\varphi$ is a homomorphism of additive groups.
Since the correspondence theorem holds for groups, the maps
$I\mapsto\varphi(I)$ and $J\mapsto\varphi^{-1}(J)$ are mutually
inverse bijections; by the two steps above, they carry ideals to
ideals.

\emph{The quotients.}
Suppose $I\leftrightarrow J$, so $J=\varphi(I)$ and
$I=\varphi^{-1}(J)$. Consider the composite
\[
  \begin{tikzcd}
    R \arrow[r,"\varphi"] \arrow[d] & S \arrow[d,"\pi"]\\
    R/I \arrow[r,dashed] & S/J
  \end{tikzcd}
\]
The composite $\pi\circ\varphi$ is surjective, and it sends $I$ to
$0$ because $\varphi(I)=J$. By the universal property, it induces a
homomorphism $R/I\to S/J$. The kernel of $\pi\circ\varphi$ is
$\varphi^{-1}(J)=I$, so the first isomorphism theorem applied to
$\pi\circ\varphi$ gives $R/I\cong S/J$.
\end{proof}

\begin{exercise}
Complete the proof that $\varphi(I)$ is an ideal of $S$ when
$\varphi:R\to S$ is surjective and $I$ is an ideal of $R$.
\end{exercise}

\section{Examples}

\begin{example}[Ideals of $\C{[t]}/(t^2-1)$]
Consider the canonical quotient map
\[
  \varphi:\C[t]\to\C[t]/(t^2-1).
\]
Every ideal of $\C[t]$ has the form $(f(t))$ for some polynomial $f$.
What are the ideals of $\C[t]/(t^2-1)$?

The map $\varphi$ is surjective with $\ker(\varphi)=(t^2-1)$.
By the correspondence theorem, ideals $J$ of $\C[t]/(t^2-1)$
correspond to ideals $I$ of $\C[t]$ containing $K=(t^2-1)$.
Writing $I=(f)$,
\[
  (f)\supseteq(t^2-1)
  \iff f\mid t^2-1=(t-1)(t+1)\text{ in }\C[t].
\]
Up to a nonzero scalar, $f$ is one of
\[
  1,\qquad t-1,\qquad t+1,\qquad (t-1)(t+1)=t^2-1.
\]
The correspondence is therefore
\[
\begin{array}{ccl}
  \text{Ideal of }\C[t] & & \text{Ideal of }\C[t]/(t^2-1)\\ \hline
  (1) & \longmapsto & (1)=\C[t]/(t^2-1)\\
  (t-1) & \longmapsto & \bigl(t-1+(t^2-1)\bigr)\\
  (t+1) & \longmapsto & \bigl(t+1+(t^2-1)\bigr)\\
  (t^2-1) & \longmapsto & 0
\end{array}
\]
% NOTE: the notes leave the images of (t-1) and (t+1) as "?"; the
% entries above are the images of the generators under phi.
\end{example}


\begin{example}[Ideals of $\Q{[x]}/(x^3-2)$]
Consider the canonical quotient map
\[
  \varphi:\Q[x]\to\Q[x]/(x^3-2).
\]
What are the ideals of $\Q[x]/(x^3-2)$? They correspond to ideals
$I=(f)$ of $\Q[x]$ with
\[
  (f)\supseteq(x^3-2)\iff f\mid x^3-2\text{ in }\Q[x].
\]
Since $x^3-2$ does not factor over $\Q$, up to a nonzero scalar
either $f=1$ or $f=x^3-2$. The correspondence is
\[
  (1)\longmapsto(1),\qquad (x^3-2)\longmapsto(0).
\]
So $\Q[x]/(x^3-2)$ has exactly two ideals, $(0)$ and $(1)$. By the
field characterization from Lecture 27, it is a field. This field
may be identified with
$\Q(\sqrt[3]{2})$.
% NOTE: p. 11 writes "Q(gamma)" without defining gamma; presumably
% gamma is a root of x^3-2, e.g. the real cube root of 2.
\end{example}

\section{Quotienting in stages}

\begin{example}\label{ex:gaussian-question}
Consider the ring $\Z[i]/(i-2)$, where
\[
  \Z[i]=\{a+bi:a,b\in\Z\},\qquad i^2=-1,
\]
is the ring of \emph{Gaussian integers}. What is this quotient?
\end{example}

Before answering, we record a general principle. Let $R$ be a ring
and $a,b\in R$. Under the quotient map $R\to R/(a)$, write
$\bar b$ for the image of $b$. Setting $\bar b=0$ in $R/(a)$ means
passing to the quotient $(R/(a))/(\bar b)$.

\begin{proposition}\label{prop:stages}
For $a,b\in R$,
\[
  \bigl(R/(a)\bigr)\big/(\bar b)\;\cong\;R/(a,b).
\]
\end{proposition}
\begin{proof}
Apply the correspondence theorem to the quotient map
$\pi:R\to R/(a)$. The ideal $(a,b)$ contains $(a)$, and its image is
\[
  \pi\bigl((a,b)\bigr)=(\bar b)\subseteq R/(a),
\]
since $\pi(ra+sb)=\bar s\,\bar b$. Hence
$R/(a,b)\cong\bigl(R/(a)\bigr)/(\bar b)$.
\end{proof}
% NOTE: the notes write (\bar b) \subset R/(0); this is a slip for R/(a).

What does this have to do with $\Z[i]/(i-2)$? The key is to
describe $\Z[i]$ itself as a quotient of a polynomial ring.

\begin{proposition}\label{prop:gaussian-presentation}
There is an isomorphism $\Z[x]/(x^2+1)\cong\Z[i]$, induced by
$x\mapsto i$.
\end{proposition}
\begin{proof}
Consider the substitution homomorphism
\[
  \varphi:\Z[x]\to\Z[i],\qquad f(x)\mapsto f(i).
\]
It is surjective, since $a+bx\mapsto a+bi$. By
Theorem~\ref{thm:first-iso}, it suffices to show that
\[
  \ker(\varphi)=\{f\in\Z[x]:f(i)=0\}=(x^2+1).
\]
Clearly $x^2+1\in\ker(\varphi)$, so $(x^2+1)\subseteq\ker(\varphi)$.

Conversely, suppose $f(x)\in\ker(\varphi)$. Since $x^2+1$ is monic,
we can divide with remainder in $\Z[x]$:
\[
  f(x)=q(x)(x^2+1)+r(x),
\]
where $r(x)=0$ or $\deg r<2$. Then
\[
  0=f(i)=q(i)\underbrace{(i^2+1)}_{=0}+r(i),
\]
so $r(i)=0$. If $r(x)\neq0$, then $r(x)=ax+b$ with $a,b\in\Z$ and
$ai+b=0$. Here $a\neq0$, since otherwise $b=0$ as well.

Why can't this happen? It would mean that $i=-b/a\in\Q$, so that
$x^2+1$ has a root in $\Q$. Then $x^2+1$ would factor into two linear
terms over $\Q$,
\[
  x^2+1=(a_1x+b_1)(a_2x+b_2),\qquad a_1,a_2,b_1,b_2\in\Q.
\]
This is impossible, since $t^2+1\geq1>0$ for every rational $t$.
Hence $r=0$ and $f\in(x^2+1)$.
\end{proof}

\begin{example}[Answer to Example~\ref{ex:gaussian-question}]
We have the commutative square of quotient maps
\[
  \begin{tikzcd}
    \Z[x] \arrow[r] \arrow[d,"x\,=\,2"'] &
      \Z[i]\cong\Z[x]/(x^2+1) \arrow[d]\\
    \Z\cong\Z[x]/(x-2) \arrow[r] & \Z[i]/(i-2)
  \end{tikzcd}
\]
The left map is the substitution $x\mapsto2$, a surjection
$\Z[x]\to\Z$ with kernel $(x-2)$. By
Propositions~\ref{prop:gaussian-presentation} and~\ref{prop:stages},
\[
  \Z[i]/(i-2)\cong\Z[x]/(x^2+1,\,x-2).
\]
Quotienting in the other order, $x-2$ first, sends $x^2+1$ to
$2^2+1=5$, so
\[
  \Z[i]/(i-2)\cong\Z[x]/(x^2+1,\,x-2)\cong\Z/(5)=\F_5.
\]
\end{example}

\section{Adjoining elements}

Given a ring $R$ and an element $\alpha$, we want a new ring
containing both $R$ and $\alpha$, constructed in a ``minimal'' way.
That ring is called $R[\alpha]$, read ``$R$ adjoin $\alpha$''.

\begin{example}
\leavevmode
\begin{itemize}
  \item Take $R=\Z$ and $\alpha=x$ a variable. Then $R[\alpha]=\Z[x]$
    is the polynomial ring.
  \item Take $R=\Z$ and $\alpha=i$, where $i^2+1=0$. Then
    \[
      \Z[i]\cong\Z[x]/(x^2+1)
    \]
    by Proposition~\ref{prop:gaussian-presentation}.
  \item Take $R=\Q$ and $\gamma=\sqrt[3]{2}$. We saw earlier that
    $\Q[x]/(x^3-2)$ is a field, and
    \[
      \Q[x]/(x^3-2)\cong\Q(\gamma)=\Q[\gamma].
    \]
\end{itemize}
\end{example}

\section{Quotients by monic polynomials}

Let $R$ be any ring, and let $f(x)\in R[x]$ be a monic polynomial,
\[
  f(x)=x^n+a_{n-1}x^{n-1}+\cdots+a_0.
\]

\begin{theorem}\label{thm:monic-quotient}
\leavevmode
\begin{enumerate}[label=(\arabic*)]
  \item The ring $R[x]/(f(x))$ has ``basis'' $1,x,x^2,\dots,x^{n-1}$
    over $R$. That is, every element of $R[x]/(f)$ has a
    representative of the form
    \[
      b_0+b_1x+\cdots+b_{n-1}x^{n-1},\qquad b_i\in R,
    \]
    and the coefficients $b_i$ are uniquely determined.
  \item Addition is done in ``vector form'':
    \[
      \bigl(b_0+\cdots+b_{n-1}x^{n-1}\bigr)
      +\bigl(c_0+\cdots+c_{n-1}x^{n-1}\bigr)
      =(b_0+c_0)+\cdots+(b_{n-1}+c_{n-1})x^{n-1}.
    \]
  \item For multiplication, note that in $R[x]/(f)$ we have
    \[
      x^n=-a_{n-1}x^{n-1}-\cdots-a_0.
    \]
    Multiply as usual, and use this relation to reduce the answer to
    a sum of terms of degree less than $n$.
\end{enumerate}
\end{theorem}

\begin{proof}[Proof of \textup{(1)}]
Let $g(x)\in R[x]$. Since $f(x)$ is monic, we can divide:
\[
  g(x)=q(x)f(x)+r(x),\qquad r(x)=0\ \text{or}\ \deg r<n.
\]
Thus in $R[x]/(f)$ we have $g(x)+(f)=r(x)+(f)$, where
$r(x)=b_0+b_1x+\cdots+b_{n-1}x^{n-1}$. So every element of
$R[x]/(f)$ has a representative of this form.

For uniqueness, suppose that $g(x)+(f)=r(x)+(f)=\tilde r(x)+(f)$
with $r$ and $\tilde r$ both zero or of degree less than $n$. Then
$r(x)-\tilde r(x)\in(f)$, and $r-\tilde r$ is zero or has degree
less than $n$. By the claim below, every nonzero polynomial in $(f)$
has degree at least $n$. Hence $r=\tilde r$.

\emph{Claim.} If $f$ is monic of degree $n$, every nonzero polynomial
in $(f)$ has degree $\geq n$. Indeed, an element of $(f)$ has the form
$h(x)f(x)$. If $h\neq0$ has leading coefficient $c$ and degree $m$,
then the leading term of $hf$ is $c\cdot1\cdot x^{m+n}$, which is
nonzero because $f$ has leading coefficient $1$. So
$\deg(hf)=m+n\geq n$.
\end{proof}

\begin{warningbox}
The claim fails if $f$ is not monic. For example, work in
$(\Z/6\Z)[x]$ with
\[
  f=2x^3+3x^2.
\]
Then $3\cdot f\in(f)$, but
\[
  3f=6x^3+9x^2=3x^2\quad\text{in }(\Z/6\Z)[x],
\]
a nonzero element of $(f)$ of degree $2<3$.
\end{warningbox}
